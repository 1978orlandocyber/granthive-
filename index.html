Swift Integration Code

Drop these two files into your Xcode project to wire up both features in GrantHive.

File 1 — GrantsGovService.swift

Fetches live grants from the Grants.gov API. Add to your Services/ group.

// GrantsGovService.swift — Add to GrantHive Xcode project import Foundation struct FederalGrant: Identifiable, Codable { let id: String let title: String let agency: String let description: String let awardCeiling: Int? let closeDate: String? let opportunityNumber: String enum CodingKeys: String, CodingKey { case id = "id" case title = "opportunityTitle" case agency = "agencyName" case description = "synopsis" case awardCeiling case closeDate case opportunityNumber } } class GrantsGovService: ObservableObject { @Published var grants: [FederalGrant] = [] @Published var isLoading = false @Published var error: String? func search(keywords: String, eligibility: String? = nil, limit: Int = 10) { isLoading = true error = nil var components = URLComponents(string: "https://api.grants.gov/v2/opportunities/search")! var queryItems = [URLQueryItem]() queryItems.append(URLQueryItem(name: "keyword", value: keywords)) queryItems.append(URLQueryItem(name: "rows", value: String(limit))) queryItems.append(URLQueryItem(name: "oppStatuses", value: "forecasted|posted")) if let elig = eligibility { queryItems.append(URLQueryItem(name: "eligibilities", value: elig)) } components.queryItems = queryItems guard let url = components.url else { return } URLSession.shared.dataTask(with: url) { [weak self] data, _, err in DispatchQueue.main.async { self?.isLoading = false guard let data = data, err == nil else { self?.error = "Network error. Try again." return } do { let decoded = try JSONDecoder().decode(GrantsResponse.self, from: data) self?.grants = decoded.hits.hits.map { $0.source } } catch { self?.error = "Parse error: \(error.localizedDescription)" } } }.resume() } } // Response wrappers struct GrantsResponse: Codable { let hits: HitsWrapper } struct HitsWrapper: Codable { let hits: [HitItem] } struct HitItem: Codable { let source: FederalGrant enum CodingKeys: String, CodingKey { case source = "_source" } }
File 2 — EligibilityService.swift

Calls the secure GrantHive backend function. No Anthropic key is stored or shipped in the app.

// EligibilityService.swift — replace YOUR-DOMAIN after deployment import Foundation struct EligibilityCheck: Codable { let requirement: String let status: String let note: String } struct EligibilityResult: Codable { let score: Int let verdict: String let summary: String let checks: [EligibilityCheck] } struct EligibilityRequest: Codable { let profile: OrganizationProfile let grantName: String let grantDescription: String } struct OrganizationProfile: Codable { let orgType: String let stage: String let industry: String let location: String let employees: String } enum EligibilityServiceError: Error { case invalidURL case invalidResponse case serverError } final class EligibilityService { // Replace this URL after Netlify deployment. private let endpoint = "https://YOUR-DOMAIN.netlify.app/.netlify/functions/check-eligibility" func check(requestBody: EligibilityRequest) async throws -> EligibilityResult { guard let url = URL(string: endpoint) else { throw EligibilityServiceError.invalidURL } var request = URLRequest(url: url) request.httpMethod = "POST" request.setValue("application/json", forHTTPHeaderField: "Content-Type") request.httpBody = try JSONEncoder().encode(requestBody) let (data, response) = try await URLSession.shared.data(for: request) guard let http = response as? HTTPURLResponse else { throw EligibilityServiceError.invalidResponse } guard (200...299).contains(http.statusCode) else { throw EligibilityServiceError.serverError } return try JSONDecoder().decode(EligibilityResult.self, from: data) } }
