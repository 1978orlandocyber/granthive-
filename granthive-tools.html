File 1 — GrantsGovService.swift

Fetches live grants from the Grants.gov API. Add to your Services/ group.

// GrantsGovService.swift — Add to GrantHive Xcode project import Foundation struct FederalGrant: Identifiable, Codable { let id: String let title: String let agency: String let description: String let awardCeiling: Int? let closeDate: String? let opportunityNumber: String enum CodingKeys: String, CodingKey { case id = "id" case title = "opportunityTitle" case agency = "agencyName" case description = "synopsis" case awardCeiling case closeDate case opportunityNumber } } class GrantsGovService: ObservableObject { @Published var grants: [FederalGrant] = [] @Published var isLoading = false @Published var error: String? func search(keywords: String, eligibility: String? = nil, limit: Int = 10) { isLoading = true error = nil var components = URLComponents(string: "https://api.grants.gov/v2/opportunities/search")! var queryItems = [URLQueryItem]() queryItems.append(URLQueryItem(name: "keyword", value: keywords)) queryItems.append(URLQueryItem(name: "rows", value: String(limit))) queryItems.append(URLQueryItem(name: "oppStatuses", value: "forecasted|posted")) if let elig = eligibility { queryItems.append(URLQueryItem(name: "eligibilities", value: elig)) } components.queryItems = queryItems guard let url = components.url else { return } URLSession.shared.dataTask(with: url) { [weak self] data, _, err in DispatchQueue.main.async { self?.isLoading = false guard let data = data, err == nil else { self?.error = "Network error. Try again." return } do { let decoded = try JSONDecoder().decode(GrantsResponse.self, from: data) self?.grants = decoded.hits.hits.map { $0.source } } catch { self?.error = "Parse error: \(error.localizedDescription)" } } }.resume() } } // Response wrappers struct GrantsResponse: Codable { let hits: HitsWrapper } struct HitsWrapper: Codable { let hits: [HitItem] } struct HitItem: Codable { let source: FederalGrant enum CodingKeys: String, CodingKey { case source = "_source" } }
File 2 — EligibilityService.swift

Sends org profile + grant description to Claude API and returns structured eligibility analysis.

// EligibilityService.swift — Add to GrantHive Xcode project import Foundation struct EligibilityResult { let score: Int // 0–100 let verdict: String // "Likely Eligible" | "Review Required" | "Ineligible" let checks: [EligCheck] let summary: String } struct EligCheck { let requirement: String let status: String // "yes" | "no" | "maybe" let note: String } class EligibilityService { private let apiKey = "YOUR_ANTHROPIC_API_KEY" func check(profile: UserProfile, grantDescription: String, completion: @escaping (EligibilityResult?) -> Void) { let prompt = buildPrompt(profile: profile, grant: grantDescription) let body: [String: Any] = [ "model": "claude-sonnet-4-20250514", "max_tokens": 1000, "messages": [["role": "user", "content": prompt]] ] var request = URLRequest(url: URL(string: "https://api.anthropic.com/v1/messages")!) request.httpMethod = "POST" request.setValue("application/json", forHTTPHeaderField: "Content-Type") request.setValue(apiKey, forHTTPHeaderField: "x-api-key") request.setValue("2023-06-01", forHTTPHeaderField: "anthropic-version") request.httpBody = try? JSONSerialization.data(withJSONObject: body) URLSession.shared.dataTask(with: request) { data, _, _ in guard let data = data, let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any], let content = (json["content"] as? [[String:Any]])?.first, let text = content["text"] as? String else { completion(nil); return } let result = self.parseResponse(text) DispatchQueue.main.async { completion(result) } }.resume() } private func buildPrompt(profile: UserProfile, grant: String) -> String { return """ You are a grant eligibility expert. Analyze this grant and tell me if the organization qualifies. Return ONLY valid JSON, no markdown. ORG PROFILE: - Type: \(profile.orgType) - Stage: \(profile.stage) - Industry: \(profile.industry) - Location: \(profile.state) - Employees: \(profile.employeeCount) GRANT DESCRIPTION: \(grant) Return this exact JSON structure: { "score": 0-100, "verdict": "Likely Eligible|Review Required|Ineligible", "summary": "2-sentence plain English summary", "checks": [ { "requirement": "requirement text", "status": "yes|no|maybe", "note": "why" } ] } """ } private func parseResponse(_ text: String) -> EligibilityResult? { guard let data = text.data(using: .utf8), let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any] else { return nil } let score = json["score"] as? Int ?? 0 let verdict = json["verdict"] as? String ?? "Unknown" let summary = json["summary"] as? String ?? "" let rawChecks = json["checks"] as? [[String: String]] ?? [] let checks = rawChecks.map { EligCheck(requirement: $0["requirement"] ?? "", status: $0["status"] ?? "maybe", note: $0["note"] ?? "") } return EligibilityResult(score: score, verdict: verdict, checks: checks, summary: summary) } }
